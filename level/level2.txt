def clock(pages, capacity):
    frames = [None] * capacity
    bits = [0] * capacity

    pointer = 0
    faults = 0

    for page in pages:

        if page in frames:
            index = frames.index(page)
            bits[index] = 1

        else:
            faults += 1

            while frames[pointer] is not None and bits[pointer] == 1:
                bits[pointer] = 0
                pointer = (pointer + 1) % capacity

            frames[pointer] = page
            bits[pointer] = 1

            pointer = (pointer + 1) % capacity

    return faults
